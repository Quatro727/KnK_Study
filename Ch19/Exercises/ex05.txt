#ifndef QUEUE_H
#define QUEUE_H

#include <stdbool.h>

#define MAX_QUEUE_SIZE 100

typedef int Item;
typedef struct {
    Item data[MAX_QUEUE_SIZE];
    int top;
} Queue;

void queue_append(Queue q, int i);
void queue_remove_head(Queue q);
int queue_head(Queue q);
int queue_last(Queue q);
bool is_empty(Queue q);

#endif
