#include <stdio.h>

#define N 10

int main(void){
    
    double ident[N][N], *p;
    int cnt = 0;

    for(p = &ident[0][0]; p < &ident[N-1][N-1]; p++){
        if(cnt % (N + 1) == 0){
            *p = 1.0;
        }
        else{
             *p= 0.0;
        }
        cnt++;
    }

    return 0;
}
