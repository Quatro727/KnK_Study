Following function didn't consider three edge cases.
1) if list is empty list
2) when new_node's value is largest value.
3) when new_node's value is lowest value in the list


struct node *insert_into_ordered_list(struct node *list, struct node *new_node){
    struct node *cur = list, *prev = NULL;
    
    //preventing null pointer access when list is null
    while(cur != NULL && cur->value <= new_node->value){
        prev = cur;
        cur = cur->next;
    }
    
    //case1: inserting first position
    if(prev == NULL) {
        new_node->next = list;
        return new_node;
    }
    
    //case2: inserting middle or last of the list
    prev->next = new_node;
    new_node->next = cur;
    
    return list;
}
