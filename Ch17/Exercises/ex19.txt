struct {
    char *cmd_name;
    void (*cmd_pointer)(void);
} file_cmd[] = {
    {"new",         new_cmd},
    {"open",        open_cmd},
    {"close",       close_cmd},
    {"close_all",   close_all_cmd},
    {"save",        save_cmd},
    {"save as",     sace_as_cmd},
    {"save all",    save_all_cmd},
    {"print",       print_cmd},
    {"exit",        exit_cmd}
};

void run_cmd(char *str){
    int i;
    int num_cmds = sizeof(file_cmd) / sizeof(file_cmd[0]);

    for(i = 0; i < num_cmds; i++){
        if(strcmp((str, file_cmd[i].cmd_name) == 0)
                return (*file_cmd[i].cmd_pointer)();
    }

    return;
}
