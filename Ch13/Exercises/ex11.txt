int strcmp(char *s, char *t){
    int i;

    for(i = 0; s[i] == t[i]; i++){
        if(s[i] =='\0') return 0;
    }
    return s[i] - t[i];
}

can be rewritten by using pointer arithmetic.

int strcmp(char *s, char *t){
    while(*s == *t){
        if(*s == '\0') return 0;
        
        s++;
        t++;
    }
    return *s - *t;
}
