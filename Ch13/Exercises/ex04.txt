(a)  skip the white-space before beginning to store input characters

int read_line(char str[], int n){
    int ch, i = 0;

    while((ch = getchar()) != '\n'){
        if(isspace(ch)) continue;

        if(i < n )str[i++] = ch;
    }
    str[i] = '\0';
    return i;
}

(c) stop reading at the first new-line character, then store the new-line character in the string 

int read_line(char str[], int n){
    int ch ,i = 0;

    while((ch = getchar() != EOF){
        if(i < n - 1) str[i++] = ch;
        if (ch == '\n') break;
    }
    str[i] = '\0';

    return i ;
}

(d) leace behind characters that it doesn't have room to store
  
int read_line(char str[], int n){
    int ch, i = 0;

    while((ch = getchar()) != '\n'){
        if(i == (n - 1)) break;
        else if(i < n) str[i++] = ch;
    }
    str[i] = '\0';

    return i;
}
